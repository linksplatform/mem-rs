2    alloc::Layout,
3    fmt::{self, Formatter},
4    marker::PhantomData,

10pub struct RawPlace<T> {
11    ptr: NonNull<T>,
12    len: usize, // use to drop at panic
13    cap: usize, // usually `cap` is same `len`
14    _marker: PhantomData<T>,
15}
16
17impl<T> RawPlace<T> {
18    pub const fn dangling() -> Self {
19        Self { ptr: NonNull::dangling(), len: 0, cap: 0, _marker: PhantomData }
20    }
21
22    pub fn cap(&self) -> usize {
23        self.cap
24    }
25
26    pub unsafe fn as_slice(&self) -> &[T] {
27        unsafe { slice::from_raw_parts(self.ptr.as_ptr(), self.len) }
28    }
29
30    pub unsafe fn as_slice_mut(&mut self) -> &mut [T] {
31        unsafe { slice::from_raw_parts_mut(self.ptr.as_ptr(), self.len) }
32    }
33
34    pub fn current_memory(&self) -> Option<(NonNull<u8>, Layout)> {
35        // rust does not support such types,
36        // so we can do better by skipping some checks and avoid an unwrap.
37        const { assert!(mem::size_of::<T>().is_multiple_of(mem::align_of::<T>())) };
38
39        if self.cap == 0 {
40            None
41        } else {
42            unsafe {
43                let layout = Layout::from_size_align_unchecked(
44                    mem::size_of::<T>().unchecked_mul(self.cap),
45                    mem::align_of::<T>(),
46                );
47                Some((self.ptr.cast(), layout))
48            }
49        }
50    }
51
52    pub unsafe fn handle_fill(
53        &mut self,
54        (ptr, cap): (NonNull<T>, usize),
55        inited: usize,
56        fill: impl FnOnce(usize, (&mut [T], &mut [MaybeUninit<T>])),
57    ) -> &mut [T] {
58        unsafe {
59            // fixme: ZST correctness isn't checked now,
60            // it forbid growing, but allow `RawPlace::<ZST>::dangling` and thus `Alloc::<ZST>::new`'s
61            const { assert!(mem::size_of::<T>() != 0) };
62
63            // Manually compute the pointer to the uninitialized part (stable alternative to slice_ptr_get + ptr_as_uninit)
64            let uninit_ptr = ptr.as_ptr().add(self.cap);
65            let uninit_len = cap - self.cap;
66            let uninit = slice::from_raw_parts_mut(uninit_ptr as *mut MaybeUninit<T>, uninit_len);
67
68            self.ptr = ptr;
69            self.cap = cap; // `ptr` and `cap` changes after panicking `fill`
70            //                 ( alloc memory )
71
72            // slice from `as_slice_mut` will be the initialized part of owned memory
73            // while (&mut [T], &mut [MaybeUninit<T>]) will be the full memory
74            fill(inited, (self.as_slice_mut(), uninit)); // panic out!
75
76            self.len = cap; // `len` is same `cap` only if `uninit` was init
77
78            crate::raw_mem::uninit::assume_init_mut(uninit)
79        }
80    }
81
82    pub fn shrink_to(&mut self, cap: usize) {
83        assert!(cap <= self.cap);
84
85        unsafe {
86            ptr::drop_in_place(&mut self.as_slice_mut()[cap..]);
87        }
88
89        self.cap = cap;
90        self.len = cap;
91    }
92
93    pub fn set_ptr(&mut self, ptr: NonNull<[u8]>) {
94        debug_assert_eq!(
95            ptr.len(),
96            self.cap * mem::size_of::<T>(),
97            "Usually you have to call `.shrink_to` first to drop shrunk memory, \
98             this should be followed by a call to `.set_ptr`."
99        );
100
101        self.ptr = ptr.cast();
102    }
103}
104
105impl<T> fmt::Debug for RawPlace<T> {
106    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
107        write!(f, "({:?}::{})", self.ptr, self.cap)
108    }
109}
